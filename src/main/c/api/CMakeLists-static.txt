# The static library of the SDK for Windows (the -static-lib archives): target_link_libraries(<app> DxFeedGraalNativeSdk)
# links it with the Windows libraries it needs (link-flags.txt), with any C runtime of the application.
cmake_minimum_required(VERSION 3.21)

# Set project
project(DxFeedGraalNativeSdk)

# Print status
message(STATUS "Imported static lib ${PROJECT_NAME}")

set(DXFG_SRC_LOCATION ${CMAKE_CURRENT_SOURCE_DIR})
set(DXFG_BIN_LOCATION ${CMAKE_CURRENT_SOURCE_DIR})

if (DEFINED GRAAL_CUSTOM_SRC_LOCATION)
    set(DXFG_SRC_LOCATION ${GRAAL_CUSTOM_SRC_LOCATION})
endif ()

if (DEFINED GRAAL_CUSTOM_BIN_LOCATION)
    set(DXFG_BIN_LOCATION ${GRAAL_CUSTOM_BIN_LOCATION})
endif ()

option(GRAAL_NATIVE_SDK_INSTALL "Prepare install" ON)

if (NOT WIN32)
    message(FATAL_ERROR "The static library of ${PROJECT_NAME} is for Windows only")
endif ()

set(DXFG_STATIC_LIBRARY "${DXFG_BIN_LOCATION}/${PROJECT_NAME}_static.lib")
# The Windows libraries that the static library needs, one per line
file(STRINGS "${DXFG_BIN_LOCATION}/link-flags.txt" DXFG_SYSTEM_LIBRARIES)

# Add imported library
add_library(${PROJECT_NAME} STATIC IMPORTED GLOBAL)

# Target properties
set_target_properties(${PROJECT_NAME}
        PROPERTIES
        INTERFACE_INCLUDE_DIRECTORIES "${DXFG_SRC_LOCATION}"
        # Path to static library
        IMPORTED_LOCATION "${DXFG_STATIC_LIBRARY}"
        INTERFACE_LINK_LIBRARIES "${DXFG_SYSTEM_LIBRARIES}"
        )

if (GRAAL_NATIVE_SDK_INSTALL)
    # Deployment (install(IMPORTED_RUNTIME_ARTIFACTS) does not take a static library)
    if (NOT "${INSTALL_STATIC_LIBRARIES}" STREQUAL "")
        install(FILES "${DXFG_STATIC_LIBRARY}" DESTINATION "${INSTALL_STATIC_LIBRARIES}" COMPONENT runtime)
    else ()
        install(FILES "${DXFG_STATIC_LIBRARY}" TYPE LIB COMPONENT runtime)
    endif ()
endif ()
